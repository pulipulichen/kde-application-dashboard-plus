cmake_minimum_required(VERSION 3.16)
project(DashboardPlus VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt5 5.15 REQUIRED COMPONENTS Quick Qml)
find_package(KF5WindowSystem 5.101 REQUIRED)
add_library(dashboardwindow STATIC native/dashboardwindow.cpp native/dashboardwindow.h)
set_target_properties(dashboardwindow PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(dashboardwindow PUBLIC Qt5::Quick KF5::WindowSystem)
target_include_directories(dashboardwindow PUBLIC native)
add_library(dashboardplusplugin MODULE native/plugin.cpp)
target_link_libraries(dashboardplusplugin PRIVATE dashboardwindow Qt5::Qml)
set_target_properties(dashboardplusplugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/dashboardplusnative")
configure_file(package/contents/ui/dashboardplusnative/qmldir ${CMAKE_BINARY_DIR}/dashboardplusnative/qmldir COPYONLY)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt5 5.15 REQUIRED COMPONENTS Test)
    add_executable(window-test tests/window-test.cpp)
    target_link_libraries(window-test PRIVATE dashboardwindow Qt5::Test Qt5::Qml)
    file(COPY package/ DESTINATION ${CMAKE_BINARY_DIR}/test-package)
    add_custom_command(TARGET dashboardplusplugin POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:dashboardplusplugin>
            ${CMAKE_BINARY_DIR}/test-package/contents/ui/dashboardplusnative/)
    add_dependencies(window-test dashboardplusplugin)
    target_compile_definitions(window-test PRIVATE NATIVE_IMPORT_DIR="${CMAKE_BINARY_DIR}/dashboardplusnative"
        DASHBOARD_QML_PATH="${CMAKE_BINARY_DIR}/test-package/contents/ui/DashboardRepresentation.qml")
    add_test(NAME window-test COMMAND window-test)
    set_tests_properties(window-test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
endif()
